CREATE EXTENSION IF NOT EXISTS "pgcrypto";

CREATE TABLE IF NOT EXISTS public.app_users (
  id TEXT PRIMARY KEY,
  username TEXT UNIQUE NOT NULL,
  full_name TEXT NOT NULL,
  role TEXT NOT NULL CHECK (role IN ('ADMIN', 'USER')),
  title TEXT,
  password_hash TEXT NOT NULL,
  cds_account TEXT,
  phone TEXT,
  avatar_url TEXT,
  is_locked BOOLEAN DEFAULT FALSE,
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS public.portfolio_capital (
  id INT PRIMARY KEY DEFAULT 1 CHECK (id = 1),
  available_cash_lkr NUMERIC(18,2) NOT NULL DEFAULT 0.00,
  total_deposited_lkr NUMERIC(18,2) NOT NULL DEFAULT 0.00,
  total_withdrawn_lkr NUMERIC(18,2) NOT NULL DEFAULT 0.00,
  updated_by TEXT,
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS public.holdings (
  symbol TEXT PRIMARY KEY,
  company_name TEXT NOT NULL,
  qty NUMERIC(18,0) NOT NULL DEFAULT 0,
  avg_price NUMERIC(18,2) NOT NULL DEFAULT 0.00,
  market_price NUMERIC(18,2) NOT NULL DEFAULT 0.00,
  updated_by TEXT,
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS public.transactions (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  tx_type TEXT NOT NULL CHECK (tx_type IN ('BUY', 'SELL', 'DEPOSIT', 'WITHDRAW')),
  symbol TEXT,
  qty NUMERIC(18,0) DEFAULT 0,
  price NUMERIC(18,2) DEFAULT 0.00,
  cse_fee_lkr NUMERIC(18,2) DEFAULT 0.00,
  total_lkr NUMERIC(18,2) NOT NULL,
  realized_pnl_lkr NUMERIC(18,2) DEFAULT 0.00,
  executed_by TEXT NOT NULL,
  notes TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS public.work_logs (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  username TEXT NOT NULL,
  work_date DATE NOT NULL,
  hours_logged NUMERIC(5,2) NOT NULL CHECK (hours_logged > 0 AND hours_logged <= 24),
  focus_area TEXT NOT NULL,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS public.tasks (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  title TEXT NOT NULL,
  assigned_to TEXT NOT NULL,
  deadline DATE NOT NULL,
  priority TEXT NOT NULL DEFAULT 'NORMAL' CHECK (priority IN ('NORMAL', 'HIGH', 'CRITICAL')),
  completed BOOLEAN DEFAULT FALSE,
  completed_by TEXT,
  completed_at TIMESTAMPTZ,
  created_by TEXT NOT NULL,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS public.vault_files (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  file_name TEXT NOT NULL,
  file_type TEXT NOT NULL,
  file_size_kb NUMERIC(10,2) NOT NULL,
  storage_path TEXT,
  note TEXT,
  uploaded_by TEXT NOT NULL,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS public.chat_messages (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  sender TEXT NOT NULL,
  role TEXT NOT NULL,
  message TEXT NOT NULL,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS public.market_events (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  title TEXT NOT NULL,
  event_date DATE NOT NULL,
  category TEXT NOT NULL,
  impact TEXT NOT NULL DEFAULT 'HIGH',
  created_by TEXT NOT NULL,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS public.journals (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  author TEXT NOT NULL,
  symbol TEXT,
  title TEXT NOT NULL,
  entry_text TEXT NOT NULL,
  admin_feedback TEXT,
  reviewed_by TEXT,
  reviewed_at TIMESTAMPTZ,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS public.emergency_state (
  id INT PRIMARY KEY DEFAULT 1 CHECK (id = 1),
  active BOOLEAN DEFAULT FALSE,
  severity TEXT DEFAULT 'CRITICAL',
  message TEXT,
  triggered_by TEXT,
  triggered_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS public.audit_logs (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  actor TEXT NOT NULL,
  role TEXT NOT NULL,
  action TEXT NOT NULL,
  detail TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW()
);
